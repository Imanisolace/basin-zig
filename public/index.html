import express from 'express'; import cors from 'cors'; import crypto from 'crypto'; import { db } from './db.js'; import { getApiKeyForUser, requireApiKey, trackUsage, checkLimit } from './usage.js'; const app = express(); const PORT = process.env.PORT || 3000; const NP_IPN_SECRET = process.env.NP_IPN_SECRET; const TREE = { items: ['alice','bob','charlie','dave'], root: '469f892beb5e486747efd9bb3aef67d8f384ae98bc5a3d54067f49f2518d2d8' }; // raw body for webhook signature check app.use('/webhook/nowpayments', express.raw({type: 'application/json'})); app.use(express.json()); app.use(cors()); // ===== HEALTH ===== app.get('/health', (req,res) => res.json({status: 'ok'})); // ===== AUTH: GET API KEY ===== app.post('/apikey', async (req,res) => { const auth = req.headers.authorization || ''; const token = auth.replace('Bearer ', ''); if(!token) return res.status(401).json({error: 'Missing Clerk token'}); // verify token with clerk here... simplified const userId = 'user_from_token'; // replace with real clerk verify const email = 'user@email.com'; const apiKey = await getApiKeyForUser(userId, email); res.json({apiKey}); }); // ===== PROVE ===== app.post('/prove', requireApiKey, checkLimit('prove'), async (req,res) => { const { item } = req.body; const idx = TREE.items.indexOf(item); if(idx === -1) return res.status(404).json({error: 'Item not found'}); const proof = [{position: 'right', data: '81b637d8fcd2c6da6359e6963113a1170de795e4b725b84d1e0b4cfd9ec58ce9'}]; await trackUsage(req.userId, 'prove'); res.json({root: TREE.root, proof, target_index: idx, valid: true}); }); // ===== VERIFY ===== app.post('/verify', requireApiKey, checkLimit('verify'), async (req,res) => { const { root, item, target_index, proof } = req.body; if(!root ||!item || target_index === undefined ||!proof) { return res.status(400).json({error: 'Missing fields'}); } const valid = root === TREE.root && TREE.items[target_index] === item; await trackUsage(req.userId, 'verify'); res.json({valid}); }); // ===== USAGE ===== app.get('/usage', async (req,res) => { const auth = req.headers.authorization || ''; const token = auth.replace('Bearer ', ''); const userId = 'user_from_token'; // replace with real clerk verify const result = await db.query('SELECT * FROM usage WHERE user_id = $1', [userId]); if(result.rows.length === 0) return res.status(404).json({error: 'No usage found. Hit /apikey first'}); const u = result.rows[0]; res.json({ email: u.email, tier: u.tier, apiKey: u.api_key, usage: {prove: u.prove_count, verify: u.verify_count}, limits: {prove: u.limit, verify: u.limit}, // fixed expires_at: u.expires_at }); }); // ===== NOWPAYMENTS WEBHOOK ===== app.post('/webhook/nowpayments', async (req,res) => { const signature = req.headers['x-nowpayments-sig']; const rawBody = req.body.toString(); const expectedSig = crypto.createHmac('sha512', NP_IPN_SECRET).update(rawBody).digest('hex'); if(signature!== expectedSig) { console.log('Invalid NP signature'); return res.status(403).send('Forbidden'); } const { payment_status, order_id } = JSON.parse(rawBody); if(payment_status === 'finished') { await db.query( `UPDATE usage SET tier = 'pro', "limit" = 999999, expires_at = NOW() + interval '30 days' WHERE user_id = $1`, [order_id] ); console.log(`✅ Upgraded ${order_id} to PRO`); } res.sendStatus(200); }); app.listen(PORT, () => console.log(`Basin API running on ${PORT}`));
