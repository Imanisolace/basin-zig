app.post('/apikey', async (req,res) => {
  const auth = req.headers.authorization || '';
  const token = auth.replace('Bearer ', '');
  if(!token) return res.status(401).json({error: 'Missing Clerk token'});

  // TEMP FOR TESTING: use headers instead of real clerk
  const userId = req.headers['x-user-id'] || 'user_from_token';
  const email = req.headers['x-email'] || 'user@email.com';

  const apiKey = await getApiKeyForUser(userId, email);
  res.json({apiKey});
});

app.post('/verify', requireApiKey, await checkLimit('verify'), async (req,res) => {
  const { root, item, target_index, proof } = req.body;
  if(!root ||!item || target_index === undefined ||!proof) {
    return res.status(400).json({error: 'Missing fields'});
  }
  const valid = root === TREE.root && TREE.items[target_index] === item;
  await trackUsage(req.userId, 'verify');
  res.json({valid});
});

app.get('/usage', async (req,res) => {
  // TEMP FOR TESTING: use headers instead of real clerk
  const userId = req.headers['x-user-id'] || 'user_from_token';
  const result = await db.query('SELECT * FROM usage WHERE user_id = $1', [userId]);
  if(result.rows.length === 0) return res.status(404).json({error: 'No usage found. Hit /apikey first'});
  const u = result.rows[0];
  res.json({
    email: u.email,
    tier: u.tier,
    apiKey: u.api_key,
    usage: {prove: u.prove_count, verify: u.verify_count},
    limits: {prove: u.prove_limit, verify: u.verify_limit}, // FIXED
    expires_at: u.expires_at
  });
});

app.post('/webhook/nowpayments', async (req,res) => {
  const signature = req.headers['x-nowpayments-sig'];
  const rawBody = req.body.toString();
  const expectedSig = crypto.createHmac('sha512', NP_IPN_SECRET).update(rawBody).digest('hex');
  if(signature!== expectedSig) {
    console.log('Invalid NP signature');
    return res.status(403).send('Forbidden');
  }
  const { payment_status, order_id } = JSON.parse(rawBody);
  if(payment_status === 'finished') {
    await db.query(
      `UPDATE usage SET tier = 'pro', prove_limit = 999999, verify_limit = 999999, expires_at = NOW() + interval '30 days' WHERE user_id = $1`,
      [order_id]
    );
    console.log(`✅ Upgraded ${order_id} to PRO`);
  }
  res.sendStatus(200);
});

app.listen(PORT, () => console.log(`Basin API running on ${PORT}`));});
